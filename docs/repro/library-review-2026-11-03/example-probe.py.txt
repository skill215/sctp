from pathlib import Path
import subprocess,time,re,json
r=Path('${REVIEW_DIR}');binary=str(r/'sctp-example');server_log=r/'logs/example-server.log';client_log=r/'logs/example-client.log';server=None
try:
 with server_log.open('w') as f:
  server=subprocess.Popen([str(r/'limited-peer')],stdout=f,stderr=subprocess.STDOUT)
  deadline=time.monotonic()+5;port=None
  while time.monotonic()<deadline:
   m=re.search(r'Listen on 127\.0\.0\.1:(\d+)',server_log.read_text())
   if m:port=m.group(1);break
   if server.poll() is not None:break
   time.sleep(0.02)
  if port is None:raise RuntimeError(server_log.read_text())
  start=time.monotonic()
  with client_log.open('w') as cf:
   client=subprocess.run([binary,'-ip','127.0.0.1','-port',port],stdout=cf,stderr=subprocess.STDOUT,timeout=25)
  log=client_log.read_text();result={'exit_code':client.returncode,'elapsed_seconds':time.monotonic()-start,'successful_writes':len(re.findall(r'write: len',log)),'last_lines':log.splitlines()[-5:]}
  (r/'example-result.json').write_text(json.dumps(result,indent=2)+'\n');print(json.dumps(result),flush=True)
finally:
 if server is not None:
  server.terminate()
  try:server.wait(timeout=3)
  except subprocess.TimeoutExpired:server.kill();server.wait()
