package sctp

import (
	"os"
	"runtime"
	"syscall"
	"testing"
)

func TestReviewFileListenerCloseReuse(t *testing.T) { reviewFileDupRace(t, false) }
func TestReviewPinnedFileDupControl(t *testing.T)   { reviewFileDupRace(t, true) }
func reviewFileDupRace(t *testing.T, pinned bool) {
	old := runtime.GOMAXPROCS(2)
	defer runtime.GOMAXPROCS(old)
	for attempt := 0; attempt < 30000; attempt++ {
		fd, err := syscall.Socket(syscall.AF_INET, syscall.SOCK_STREAM|syscall.SOCK_CLOEXEC, syscall.IPPROTO_SCTP)
		if err != nil {
			t.Fatal(err)
		}
		if err := syscall.Bind(fd, &syscall.SockaddrInet4{Addr: [4]byte{127, 0, 0, 1}}); err != nil {
			syscall.Close(fd)
			t.Fatal(err)
		}
		if err := syscall.Listen(fd, 8); err != nil {
			syscall.Close(fd)
			t.Fatal(err)
		}
		file := os.NewFile(uintptr(fd), "original-sctp-listener")
		ready := make(chan struct{})
		done := make(chan *SCTPListener, 1)
		go func() {
			close(ready)
			if !pinned {
				ln, _ := FileListener(file)
				done <- ln
				return
			}
			raw, err := file.SyscallConn()
			if err != nil {
				done <- nil
				return
			}
			dup := -1
			err = raw.Control(func(v uintptr) {
				n, _, e := syscall.Syscall(syscall.SYS_FCNTL, v, syscall.F_DUPFD_CLOEXEC, 0)
				if e == 0 {
					dup = int(n)
				}
			})
			if err != nil || dup < 0 {
				done <- nil
				return
			}
			owned, err := socketFile(dup)
			if err != nil {
				syscall.Close(dup)
				done <- nil
				return
			}
			done <- &SCTPListener{fd: dup, file: owned, io: newSocketIO(owned)}
		}()
		<-ready
		file.Close()
		replacement, err := os.Open(os.DevNull)
		if err != nil {
			t.Fatal(err)
		}
		ln := <-done
		wrong := false
		if ln != nil {
			raw, _ := ln.SyscallConn()
			var stat syscall.Stat_t
			if err := raw.Control(func(v uintptr) {
				if syscall.Fstat(int(v), &stat) != nil {
					t.Error("fstat failed")
				}
			}); err != nil {
				t.Error(err)
			}
			wrong = stat.Mode&syscall.S_IFMT != syscall.S_IFSOCK
			ln.Close()
		}
		replacement.Close()
		if wrong {
			t.Fatalf("attempt=%d pinned=%v: imported a non-socket after closing the original SCTP file", attempt, pinned)
		}
	}
	t.Logf("30000 concurrent close/reuse attempts: pinned=%v", pinned)
}
