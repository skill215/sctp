//go:build linux && !386

// Reproduction probes for three additional observations in release 0.7.1.
// Copy this file to round3_regressions_linux_test.go at the repository root.
// Run on Linux with SCTP enabled:
//   go test -race -run '^TestRound3' -v -count=5 -timeout=60s .
// The unfixed release is expected to fail the three observation checks. The
// unwrapped reader, two-address listener, concrete AcceptSCTP, and TCP checks
// are positive controls and must pass. Reuses existing lifecycle/timeout helpers.
// Verified on Linux/amd64 with Go 1.26.10 and Go 1.27.2 on 2026-10-30.

// 2026-10-30 recheck: TestRound3AcceptErrorNil is a P3 consistency check,
// not an established net.Listener contract violation or release blocker (#13).

package sctp

import (
	"errors"
	"fmt"
	"io"
	"net"
	"reflect"
	"syscall"
	"testing"
	"time"
	"unsafe"
)

func round3AbortPeer(t *testing.T, server *SCTPConn) {
	t.Helper()
	raw, err := server.SyscallConn()
	if err != nil {
		t.Fatal(err)
	}
	var optionErr error
	if err := raw.Control(func(fd uintptr) {
		optionErr = syscall.SetsockoptLinger(int(fd), syscall.SOL_SOCKET, syscall.SO_LINGER, &syscall.Linger{Onoff: 1, Linger: 0})
	}); err != nil {
		t.Fatal(err)
	}
	if optionErr != nil {
		t.Fatal(optionErr)
	}
	if err := server.Close(); err != nil {
		t.Fatal(err)
	}
}

func TestRound3AcceptErrorNil(t *testing.T) {
	ln := lifecycleListener(t)
	if err := ln.Close(); err != nil {
		t.Fatal(err)
	}
	var generic net.Listener = ln
	conn, err := generic.Accept()
	if err == nil {
		if conn != nil {
			conn.Close()
		}
		t.Fatal("expected closed listener error")
	}
	t.Logf("Accept error=%v conn interface nil=%v dynamic type=%T", err, conn == nil, conn)
	if conn != nil {
		t.Fatalf("Accept returned a non-nil net.Conn containing nil pointer: %v", reflect.ValueOf(conn).IsNil())
	}
}

func TestRound3WrappedResetReadCount(t *testing.T) {
	client, server := lifecyclePair(t)
	wrapped := NewSCTPSndRcvInfoWrappedConn(client)
	raw, _ := client.SyscallConn()
	if err := setSocketTimeout(raw, syscall.SO_RCVTIMEO, 100*time.Millisecond); err != nil {
		t.Fatal(err)
	}
	round3AbortPeer(t, server)
	buf := make([]byte, 128)
	n, err := wrapped.Read(buf)
	t.Logf("wrapped Read after reset=%d,%v", n, err)
	if !errors.Is(err, syscall.ECONNRESET) {
		t.Fatalf("reset trigger not exercised: %v", err)
	}
	if n < 0 || n > len(buf) {
		t.Fatalf("wrapped net.Conn.Read returned invalid count %d", n)
	}
}

func TestRound3ReadAllAfterReset(t *testing.T) {
	for _, wrap := range []bool{false, true} {
		name := "plain"
		if wrap {
			name = "wrapped"
		}
		t.Run(name, func(t *testing.T) {
			client, server := lifecyclePair(t)
			var reader io.Reader = client
			if wrap {
				reader = NewSCTPSndRcvInfoWrappedConn(client)
			}
			raw, _ := client.SyscallConn()
			if err := setSocketTimeout(raw, syscall.SO_RCVTIMEO, 100*time.Millisecond); err != nil {
				t.Fatal(err)
			}
			round3AbortPeer(t, server)
			defer func() {
				if p := recover(); p != nil {
					t.Errorf("io.ReadAll panicked on peer reset: %v", p)
				}
			}()
			buf, err := io.ReadAll(reader)
			t.Logf("ReadAll wrap=%v bytes=%d error=%v", wrap, len(buf), err)
			if !errors.Is(err, syscall.ECONNRESET) {
				t.Fatalf("reset trigger not exercised: %v", err)
			}
		})
	}
}

func TestRound3ManyLocalAddresses(t *testing.T) {
	for _, count := range []int{2, 300} {
		t.Run(fmt.Sprint(count), func(t *testing.T) {
			addr := &SCTPAddr{Port: 0}
			for i := 0; i < count; i++ {
				addr.IPAddrs = append(addr.IPAddrs, net.IPAddr{IP: net.IPv4(127, 0, byte(i/254), byte(i%254+1))})
			}
			ln, err := ListenSCTP("sctp4", addr)
			if err != nil {
				t.Fatalf("valid loopback multihoming setup failed: %v", err)
			}
			defer ln.Close()
			raw, err := ln.SyscallConn()
			if err != nil {
				t.Fatal(err)
			}
			var large struct {
				assocID   int32
				count     uint32
				addresses [16384]byte
			}
			size := uint32(unsafe.Sizeof(large))
			var optionErr error
			if err := raw.Control(func(fd uintptr) {
				_, _, optionErr = getsockopt(int(fd), SCTP_GET_LOCAL_ADDRS, uintptr(unsafe.Pointer(&large)), &size)
			}); err != nil {
				t.Fatal(err)
			}
			if optionErr != nil {
				t.Fatalf("large buffer control failed: %v", optionErr)
			}
			t.Logf("configured=%d kernel returned=%d bytes=%d", count, large.count, size)
			if int(large.count) != count {
				t.Fatalf("kernel did not bind the requested address list: %d", large.count)
			}
			var boundedErr error
			if err := raw.Control(func(fd uintptr) { _, boundedErr = sctpGetAddrs(int(fd), 0, SCTP_GET_LOCAL_ADDRS) }); err != nil {
				t.Fatal(err)
			}
			t.Logf("fixed-buffer address query error=%v", boundedErr)
			got := ln.Addr()
			t.Logf("public listener Addr nil=%v", got == nil)
			if got == nil {
				t.Fatal("usable listener Addr is nil although the kernel can return every bound address")
			}
			if len(got.(*SCTPAddr).IPAddrs) != count {
				t.Fatal("public address list incomplete")
			}
		})
	}
}

func TestRound3AcceptSCTPAndTCPControls(t *testing.T) {
	ln := lifecycleListener(t)
	ln.Close()
	c, err := ln.AcceptSCTP()
	if c != nil || err == nil {
		t.Fatalf("concrete SCTP Accept=%v,%v", c, err)
	}
	tcp, err := net.Listen("tcp4", "127.0.0.1:0")
	if err != nil {
		t.Fatal(err)
	}
	tcp.Close()
	cc, err := tcp.Accept()
	if cc != nil || err == nil {
		t.Fatalf("standard interface Accept=%v,%v", cc, err)
	}
}
